```sql
CREATE TABLE alembic_version (
	version_num VARCHAR(32) NOT NULL, 
	CONSTRAINT alembic_version_pkc PRIMARY KEY (version_num)
);
CREATE TABLE postings_raw (
	id INTEGER NOT NULL, 
	source TEXT NOT NULL, 
	posting_id TEXT NOT NULL, 
	payload_json TEXT NOT NULL, 
	fetched_at DATETIME NOT NULL, 
	period_from DATE, 
	period_to DATE, 
	PRIMARY KEY (id), 
	UNIQUE (source, posting_id)
);
CREATE TABLE postings_flat (
	source TEXT NOT NULL, 
	posting_id TEXT NOT NULL, 
	sku TEXT NOT NULL, 
	qty INTEGER NOT NULL, 
	price NUMERIC NOT NULL, 
	discount_mp NUMERIC, 
	final_price NUMERIC NOT NULL, 
	posting_date DATE NOT NULL, 
	UNIQUE (source, posting_id, sku)
);
CREATE TABLE etl_watermarks (
	source TEXT NOT NULL, 
	last_successful_iso DATETIME NOT NULL, 
	PRIMARY KEY (source)
);
CREATE TABLE jobs (
	id INTEGER NOT NULL, 
	name TEXT NOT NULL, 
	params_json TEXT, 
	status TEXT NOT NULL, 
	started_at DATETIME, 
	finished_at DATETIME, 
	result_json TEXT, 
	retries INTEGER, 
	PRIMARY KEY (id), 
	CHECK (status IN ('NEW','RUNNING','DONE','RETRYABLE_ERROR','FAILED'))
);
CREATE TABLE settings (
	"key" TEXT NOT NULL, 
	value TEXT NOT NULL, 
	PRIMARY KEY ("key")
);
CREATE TABLE agent_events (
	id INTEGER NOT NULL, 
	ts DATETIME NOT NULL, 
	agent_role TEXT NOT NULL, 
	task_id TEXT, 
	event TEXT NOT NULL, 
	details_json TEXT NOT NULL, 
	PRIMARY KEY (id)
);
CREATE TABLE doc_registry (
	doc_name TEXT NOT NULL, 
	version TEXT NOT NULL, 
	content_hash TEXT NOT NULL, 
	updated_at DATETIME NOT NULL, 
	PRIMARY KEY (doc_name)
);
CREATE TABLE change_log (
	id INTEGER NOT NULL, 
	ts DATETIME NOT NULL, 
	agent_role TEXT NOT NULL, 
	artifact_manifest_json TEXT NOT NULL, 
	commit_id TEXT, 
	related_docs TEXT, 
	PRIMARY KEY (id)
);
CREATE TABLE code_registry (
	id INTEGER NOT NULL, 
	file_path TEXT NOT NULL, 
	sha256 TEXT NOT NULL, 
	indexed_at DATETIME NOT NULL, 
	PRIMARY KEY (id), 
	UNIQUE (file_path)
);
CREATE TABLE symbol_index (
	id INTEGER NOT NULL, 
	file_path TEXT NOT NULL, 
	symbol_name TEXT NOT NULL, 
	symbol_type TEXT NOT NULL, 
	line_start INTEGER NOT NULL, 
	line_end INTEGER NOT NULL, 
	PRIMARY KEY (id), 
	FOREIGN KEY(file_path) REFERENCES code_registry (file_path)
);
CREATE TABLE call_graph_edges (
	id INTEGER NOT NULL, 
	source_symbol TEXT NOT NULL, 
	target_symbol TEXT NOT NULL, 
	file_path TEXT NOT NULL, 
	line_number INTEGER NOT NULL, 
	PRIMARY KEY (id), 
	FOREIGN KEY(file_path) REFERENCES code_registry (file_path)
);
CREATE TABLE features (
	id INTEGER NOT NULL, 
	title TEXT NOT NULL, 
	intent_json TEXT, 
	status TEXT NOT NULL, 
	priority INTEGER, 
	created_at DATETIME NOT NULL, 
	created_by TEXT, 
	env TEXT, 
	PRIMARY KEY (id), 
	CHECK (status IN ('NEW','PLANNED','RUNNING','DONE','FAILED'))
);
CREATE TABLE graph_runs (
	run_id TEXT NOT NULL, 
	feature_id INTEGER NOT NULL, 
	graph_name TEXT NOT NULL, 
	thread_id TEXT, 
	state_json TEXT, 
	status TEXT, 
	last_checkpoint_at DATETIME, env TEXT, 
	PRIMARY KEY (run_id), 
	FOREIGN KEY(feature_id) REFERENCES features (id)
);
CREATE INDEX idx_features_status ON features (status);
CREATE INDEX idx_features_priority ON features (priority);
CREATE INDEX idx_graph_runs_feature_id ON graph_runs (feature_id);
CREATE INDEX idx_graph_runs_status ON graph_runs (status);
CREATE TABLE token_stats (
                        id INTEGER PRIMARY KEY AUTOINCREMENT,
                        date DATE NOT NULL,
                        agent_role TEXT NOT NULL,
                        model TEXT NOT NULL,
                        input_tokens INTEGER NOT NULL DEFAULT 0,
                        output_tokens INTEGER NOT NULL DEFAULT 0,
                        total_calls INTEGER NOT NULL DEFAULT 0,
                        UNIQUE(date, agent_role, model)
                    );
CREATE TABLE sqlite_sequence(name,seq);
CREATE TABLE IF NOT EXISTS "tasks" (id INTEGER NOT NULL, feature_id INTEGER NOT NULL, role TEXT NOT NULL, dsl_json TEXT, status TEXT NOT NULL, attempts INTEGER, budget_tokens INTEGER, scheduled_at DATETIME, started_at DATETIME, finished_at DATETIME, PRIMARY KEY (id), FOREIGN KEY(feature_id) REFERENCES features (id), CHECK (status IN ('NEW','RUNNING','DONE','RETRYABLE_ERROR','FAILED','WAIT_BUDGET')));
CREATE UNIQUE INDEX idx_graph_runs_active_unique ON graph_runs (feature_id, env) WHERE status IN ('NEW', 'RUNNING', 'PENDING');
CREATE INDEX idx_graph_runs_feat_env_status ON graph_runs (feature_id, env, status);
```
